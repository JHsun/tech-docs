2.3.4 產生 vmlinux.h

這是 CO-RE 程式必備的 header，由 bpftool 從 Kernel 的 BTF 資訊生成。內容包含當前 Kernel 全部 type 定義（struct task_struct、struct sock 等）。

terminal
# 在你的專案 headers 目錄產生
mkdir -p ~/work/headers
sudo bpftool btf dump file /sys/kernel/btf/vmlinux format c > ~/work/headers/vmlinux.h

# 檢查產生的檔案大小（通常 5-20 MB 文字檔）
ls -lh ~/work/headers/vmlinux.h
wc -l ~/work/headers/vmlinux.h
# > 100,000 行不誇張
為何需要 vmlinux.h

傳統 eBPF 程式得 #include <linux/sched.h> 等 Kernel header；但這些 header 只能對應「當前正在編譯主機的 Kernel」。CO-RE 改用 vmlinux.h（由 BTF 生成），加上 clang 編譯時記錄欄位的邏輯位置而不是實體偏移量。執行時 libbpf / cilium-ebpf / aya 會根據目標 Kernel 的 BTF 動態 relocate，達成「一次編譯、處處運行」。詳見冊四第 15 章。

2.6 第一支程式：tracepoint Hello World

2.6.1 我們要做什麼

第一支 eBPF 程式選擇 tracepoint（不是 XDP），原因有三：

  1. 不需要網卡介面：在任何 Linux 機器都能跑，包括 VM、雲端、容器內。
  2. 立刻看得到效果：tracepoint 觀察 syscalls:sys_enter_execve，每次任何人在系統內執行命令時都會觸發。
  3. 結構最簡潔：不需要處理封包，專注在「掛載 + 計數 + 輸出」這三件事。

程式行為：

每次系統內有人呼叫 execve()（也就是執行任何新程式）時，eBPF 程式會被觸發、把 PID 與執行的命令名稱（comm）送到 user space，user space 程式把訊息印出來。

sequenceDiagram participant U as User
(在 shell 打 ls) participant K as Kernel participant TP as tracepoint
sys_enter_execve participant EBPF as eBPF Program participant RB as Ring Buffer participant APP as User Space App U->>K: execve("ls") K->>TP: 觸發 tracepoint TP->>EBPF: 呼叫 BPF 程式 EBPF->>EBPF: 讀取 PID & comm EBPF->>RB: bpf_ringbuf_submit(event) Note over EBPF: BPF 結束，return 0 K->>U: 繼續執行 ls RB->>APP: poll 取得事件 APP->>APP: 印出 "PID 1234: ls"

bpf/hello.c
// SPDX-License-Identifier: GPL-2.0
//
// 第一支 eBPF 程式：tracepoint sys_enter_execve
// 觀察任何 process 啟動，把 PID 與命令名透過 ring buffer 送到 user space。

#include "headers/vmlinux.h"
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

char LICENSE[] SEC("license") = "GPL";

// User space 收到的事件結構
struct event {
    __u32 pid;
    __u8  comm[16];
};

// Ring Buffer Map：BPF 寫入、user space 讀出
struct {
    __uint(type, BPF_MAP_TYPE_RINGBUF);
    __uint(max_entries, 256 * 1024);  // 256 KiB
} events SEC(".maps");

// 為了讓 bpf2go 自動產生 Go struct，宣告一個 dummy
const struct event *unused __attribute__((unused));

// tracepoint 程式：syscalls/sys_enter_execve
SEC("tracepoint/syscalls/sys_enter_execve")
int handle_execve(void *ctx) {
    // 從 ring buffer 預留一個 event 大小的空間
    struct event *e = bpf_ringbuf_reserve(&events, sizeof(*e), 0);
    if (!e) {
        return 0;  // ring buffer 滿，丟棄這次事件
    }

    // 取目前 task 的 PID 與 comm
    e->pid = bpf_get_current_pid_tgid() >> 32;
    bpf_get_current_comm(&e->comm, sizeof(e->comm));

    // 提交事件
    bpf_ringbuf_submit(e, 0);
    return 0;
}

package main

// 這個指令會由 `go generate` 觸發 bpf2go：
//   1. 用 clang 把 bpf/hello.c 編譯成 BPF bytecode
//   2. 產生 hello_bpfel.go / hello_bpfeb.go（embed 編譯結果 + Go scaffolding）

//go:generate go tool bpf2go -cc clang -target amd64,arm64 \
//  -type event hello bpf/hello.c -- -I./bpf/headers
main.go
package main

import (
    "bytes"
    "encoding/binary"
    "errors"
    "log"
    "os"
    "os/signal"
    "syscall"

    "github.com/cilium/ebpf/link"
    "github.com/cilium/ebpf/ringbuf"
    "github.com/cilium/ebpf/rlimit"
)

func main() {
    // 1. 在 5.11 之前的 Kernel 需要解除 RLIMIT_MEMLOCK
    if err := rlimit.RemoveMemlock(); err != nil {
        log.Fatalf("RemoveMemlock: %v", err)
    }

    // 2. 載入 bpf2go 產生的 spec 與 objects
    objs := helloObjects{}
    if err := loadHelloObjects(&objs, nil); err != nil {
        log.Fatalf("loading objects: %v", err)
    }
    defer objs.Close()

    // 3. 把 BPF 程式 attach 到 tracepoint
    tp, err := link.Tracepoint("syscalls", "sys_enter_execve", objs.HandleExecve, nil)
    if err != nil {
        log.Fatalf("opening tracepoint: %v", err)
    }
    defer tp.Close()

    // 4. 開啟 ring buffer reader
    rd, err := ringbuf.NewReader(objs.Events)
    if err != nil {
        log.Fatalf("opening ringbuf reader: %v", err)
    }
    defer rd.Close()

    // 5. 處理 Ctrl-C：關閉 reader 解開 blocking read
    stopper := make(chan os.Signal, 1)
    signal.Notify(stopper, os.Interrupt, syscall.SIGTERM)
    go func() {
        <-stopper
        if err := rd.Close(); err != nil {
            log.Fatalf("closing ringbuf reader: %v", err)
        }
    }()

    log.Println("Listening for events.. (Ctrl-C 結束)")

    // 6. 主迴圈：讀取事件並印出
    var event helloEvent
    for {
        record, err := rd.Read()
        if err != nil {
            if errors.Is(err, ringbuf.ErrClosed) {
                log.Println("Received signal, exiting..")
                return
            }
            log.Printf("reading from reader: %v", err)
            continue
        }

        // 把 raw bytes parse 為 helloEvent struct
        if err := binary.Read(
            bytes.NewBuffer(record.RawSample),
            binary.LittleEndian, &event,
        ); err != nil {
            log.Printf("parsing ringbuf event: %v", err)
            continue
        }

        comm := bytes.TrimRight(event.Comm[:], "\x00")
        log.Printf("PID=%-7d comm=%s", event.Pid, comm)
    }
}

#![no_std]
#![no_main]

use aya_ebpf::{
    helpers::{bpf_get_current_comm, bpf_get_current_pid_tgid},
    macros::{map, tracepoint},
    maps::RingBuf,
    programs::TracePointContext,
};
use hello_ebpf_rs_common::Event;

// 256 KiB ring buffer
#[map]
static EVENTS: RingBuf = RingBuf::with_byte_size(256 * 1024, 0);

#[tracepoint]
pub fn handle_execve(ctx: TracePointContext) -> u32 {
    match try_handle_execve(ctx) {
        Ok(_) => 0,
        Err(_) => 1,
    }
}

fn try_handle_execve(_ctx: TracePointContext) -> Result<(), i64> {
    // 從 ring buffer 預留 entry
    let mut entry = EVENTS.reserve::<Event>(0).ok_or(1_i64)?;

    // 透過 entry.write() 拿到可寫的引用
    let event = unsafe { &mut *entry.as_mut_ptr() };

    // PID 在 pid_tgid 的高 32 bits
    event.pid = (bpf_get_current_pid_tgid() >> 32) as u32;

    // 取得 comm（process 名稱）
    let comm = bpf_get_current_comm().map_err(|_| 1_i64)?;
    event.comm = comm;

    entry.submit(0);
    Ok(())
}

hello-ebpf-rs/src/main.rs
use aya::{
    maps::RingBuf,
    programs::TracePoint,
    Ebpf,
};
use aya_log::EbpfLogger;
use hello_ebpf_rs_common::Event;
use log::{info, warn};
use std::convert::TryInto;
use tokio::{io::unix::AsyncFd, signal};

#[tokio::main]
async fn main() -> Result<(), anyhow::Error> {
    env_logger::init();

    // 1. 載入由 BPF crate 編譯出的 ELF（include_bytes! 內嵌至 user binary）
    #[cfg(debug_assertions)]
    let mut bpf = Ebpf::load(include_bytes_aligned!(
        "../../target/bpfel-unknown-none/debug/hello-ebpf-rs"
    ))?;
    #[cfg(not(debug_assertions))]
    let mut bpf = Ebpf::load(include_bytes_aligned!(
        "../../target/bpfel-unknown-none/release/hello-ebpf-rs"
    ))?;

    if let Err(e) = EbpfLogger::init(&mut bpf) {
        warn!("aya log init failed: {}", e);
    }

    // 2. 取出 BPF program 並 attach 到 tracepoint
    let program: &mut TracePoint = bpf
        .program_mut("handle_execve")
        .unwrap()
        .try_into()?;
    program.load()?;
    program.attach("syscalls", "sys_enter_execve")?;

    // 3. 開啟 ring buffer
    let mut ring_buf = RingBuf::try_from(bpf.take_map("EVENTS").unwrap())?;
    let mut async_fd = AsyncFd::new(ring_buf)?;

    info!("Listening for events.. (Ctrl-C 結束)");

    // 4. 主迴圈：以 async + select 等待 ringbuf 或 Ctrl-C
    loop {
        tokio::select! {
            _ = signal::ctrl_c() => {
                info!("收到中斷訊號，結束。");
                break;
            }
            guard = async_fd.readable_mut() => {
                let mut guard = guard?;
                let rb = guard.get_inner_mut();
                while let Some(item) = rb.next() {
                    let bytes = item.as_ref();
                    if bytes.len() >= std::mem::size_of::<Event>() {
                        let event = unsafe {
                            std::ptr::read_unaligned(bytes.as_ptr() as *const Event)
                        };
                        let comm = std::str::from_utf8(&event.comm)
                            .unwrap_or("?")
                            .trim_end_matches('\0');
                        info!("PID={:<7} comm={}", event.pid, comm);
                    }
                }
                guard.clear_ready();
            }
        }
    }
    Ok(())
}

不論你選 Go 或 Rust，跑起來的效果一樣：每次系統內有 process 啟動，都會看到一行 PID=xxx comm=yyy。差別在 user space 程式語言、專案組織方式，以及 BPF 端是用 C（Go 路線）還是 Rust（aya 路線）。

2.7 程式生命週期解剖

剛才的 Hello World 表面上很簡單，但底層發生了很多事。下面這張圖把從 go run / cargo run 到事件列印的完整流程拆開：

sequenceDiagram participant DEV as 開發者
(終端機) participant TC as 編譯工具鏈
clang / bpf-linker participant USER as User Space 程式
(Go / Rust 二進位) participant SYS as bpf() syscall participant VRF as Verifier participant JIT as JIT participant HOOK as Tracepoint hook participant RB as Ring Buffer DEV->>TC: go generate / cargo xtask TC->>TC: clang 編 C/Rust 為
BPF bytecode (.o) TC-->>USER: 二進位嵌入 .o DEV->>USER: 執行 USER->>SYS: BPF_PROG_LOAD
(載入 program) SYS->>VRF: 驗證程式 VRF-->>SYS: PASS SYS->>JIT: 編成原生機器碼 JIT-->>SYS: 完成 SYS-->>USER: 回傳 program FD USER->>SYS: BPF_LINK_CREATE
(attach tracepoint) SYS->>HOOK: 註冊 BPF 程式 USER->>USER: 進入 ring buffer 主迴圈 Note over HOOK,RB: 系統內有 process
呼叫 execve() HOOK->>HOOK: BPF 程式被執行 HOOK->>RB: bpf_ringbuf_submit RB-->>USER: epoll 通知 USER->>USER: 印出 PID + comm

2.7.1 載入階段（一次性）

  1. 讀取 .o 物件檔：cilium/ebpf 的 helloObjects 或 aya 的 Ebpf::load 從 ELF 解析出 program、map、BTF。
  2. 建立 Map：透過 BPF_MAP_CREATE syscall 在 Kernel 建立 ring buffer。
  3. 載入 Program：透過 BPF_PROG_LOAD 把 bytecode 送進 Kernel，Kernel 內部會：
    • 跑 Verifier 靜態分析。
    • 對 CO-RE 做 relocation（依目標 Kernel BTF 修正欄位偏移）。
    • 用 JIT 編成原生指令。
  4. Attach 到 hook：透過 BPF_LINK_CREATE 把 program 綁到 tracepoint。回傳的 link FD 是程式存活的「錨點」——一旦關閉這個 FD，hook 就會被解開。

2.7.2 運行階段（每次事件）

這時 user space 程式只剩下做一件事：從 ring buffer 讀資料。所有的工作（觸發、計算、輸出）都在 Kernel 中由 BPF 程式完成：

flowchart LR EXEC["execve syscall"] TP["tracepoint
sys_enter_execve"] EBPF["BPF 程式
handle_execve"] HELP1["bpf_get_current_pid_tgid"] HELP2["bpf_get_current_comm"] RES["bpf_ringbuf_reserve"] SUB["bpf_ringbuf_submit"] APP["User App
(epoll)"] EXEC --> TP --> EBPF EBPF --> RES EBPF --> HELP1 EBPF --> HELP2 EBPF --> SUB --> APP

2.7.3 結束階段

當 user space 程式收到 Ctrl-C：

  1. 關閉 ring buffer reader（解開阻塞）。
  2. defer tp.Close()（Go）或 Rust 的 Drop trait 自動關閉 link。
  3. 關閉 program FD、map FD。
  4. Kernel 偵測到所有 reference 歸零，自動清除該 program 與 map。
為何 user space 程式不能退出後 BPF 程式繼續跑？

預設情況下，BPF program 與 map 的生命週期綁在 user space 程式手上。一旦 process 退出、所有 FD 關閉，Kernel 就會回收。如果你希望 BPF 程式存活到 user space 程式退出之後（例如背景 daemon 已經部署完，loader 程式可以離場），可以用 Pinning 機制把 program / map 釘到 /sys/fs/bpf/ 下。詳見冊二第 7 章。

2.8.2 argument list too long / verifier rejected program

Verifier 拒絕載入。常見原因：

  • 程式超過 1M 指令上限（Kernel 5.2 後放寬到 1M）。
  • 讀取了未檢查 NULL 的 pointer。
  • 迴圈邊界 Verifier 無法靜態推導。

排錯第一步：看 Verifier log。Go 端：

2.9 本章小結與下一步

2.9.1 本章重點回顧

  1. 環境檢查清單：Kernel 5.15+、BTF 啟用、bpf 檔案系統掛載、必要工具鏈安裝。
  2. 共通工具鏈：clang / LLVM / libbpf / bpftool / vmlinux.h，不論用 Go 還是 Rust 都需要。
  3. Go 路線（cilium/ebpf）：BPF 端用 C、user space 用 Go，bpf2go 自動產生 scaffolding。
  4. Rust 路線（aya）：BPF 端與 user space 都用 Rust，cargo-generate 提供完整 workspace 範本。
  5. 第一支程式：tracepoint sys_enter_execve + ring buffer，每次系統內 process 啟動都會輸出 PID + comm。
  6. 程式生命週期：載入 → Verifier → JIT → attach → 運行 → 退出時自動清除（除非 pinning）。
