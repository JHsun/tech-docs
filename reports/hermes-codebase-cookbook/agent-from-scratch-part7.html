Cron Scheduler 架構

Cron scheduler 是一個簡單的 tick loop：每分鐘喚醒一次， 掃描 job store，把到期的 job 交給 AIAgent 執行（或直接執行腳本）， 把結果投遞到指定目標。

圖 7-1 · Cron Scheduler Tick Loop
flowchart LR TICK["每分鐘喚醒"] --> SCAN["掃描 job store\n找到期任務"] SCAN --> DUE{"有到期\n任務?"} DUE -->|"否"| SLEEP["睡眠到下分鐘"] SLEEP --> TICK DUE -->|"是"| TYPE{"任務類型?"} TYPE -->|"no-agent"| SCRIPT["直接執行 Python 腳本\n零 LLM 費用"] TYPE -->|"agent"| AGENT["建立 AIAgent\n執行 prompt"] SCRIPT --> DELIVER["投遞結果\n到指定目標"] AGENT --> DELIVER DELIVER --> SLEEP

Cron 架構的三個組件

Job Store（cron/job_store.py）

以 JSON 檔儲存所有 job 定義：schedule、prompt、no_agent flag、script、delivery target。提供 CRUD 和「找到期 job」查詢。

Scheduler（cron/scheduler.py）

Tick loop 主程序。用 asyncio 非同步，每 60 秒喚醒一次，呼叫 job store 找到期任務，分發給執行器。

Cron Tool（tools/cron_tool.py）

agent 在對話中用來管理 cron job 的 tool：create / list / pause / resume / delete。參數包含自然語言 schedule。

Job Store 設計與 Cron Tool

pythoncron/job_store.py（完整實作）
"""
cron/job_store.py — Cron Job Store
以 JSON 檔儲存所有 cron job，提供 CRUD 和 next-due 查詢。
"""
from __future__ import annotations
import json, time, uuid
from dataclasses import dataclass, asdict
from pathlib import Path
from agent.prompt_builder import get_agent_home


@dataclass
class CronJob:
    """單一 cron job 的完整定義。"""
    id: str
    name: str
    schedule: str         # cron expression 或 natural language
    schedule_seconds: int  # 解析後的間隔秒數（簡化版）
    prompt: str            # agent 執行的 prompt
    no_agent: bool         # True = 純腳本模式
    script: str            # no_agent=True 時執行的 Python 腳本
    enabled: bool
    last_run: float        # Unix timestamp
    next_run: float        # Unix timestamp
    created_at: float
    toolsets: list[str]   # agent 模式時啟用的 toolset
    workdir: str           # 執行目錄
    run_count: int = 0


def _parse_schedule(schedule: str) -> int:
    """
    把 schedule 字串解析成間隔秒數（簡化版）。
    完整實作應支援 cron expression 和自然語言解析。
    """
    schedule = schedule.lower().strip()
    # 常用間隔
    presets = {
        "every 1h": 3600, "every hour": 3600, "hourly": 3600,
        "every 30m": 1800, "every 2h": 7200, "every 6h": 21600,
        "every 12h": 43200, "daily": 86400, "every day": 86400,
        "weekly": 604800,
    }
    if schedule in presets:
        return presets[schedule]
    # 簡單的 "every Nm" 或 "every Ns" 解析
    import re
    m = re.match(r'every\s+(\d+)\s*(m|min|h|hour|d|day|s|sec)', schedule)
    if m:
        n, unit = int(m.group(1)), m.group(2)
        mult = {'s': 1, 'sec': 1, 'm': 60, 'min': 60, 'h': 3600, 'hour': 3600, 'd': 86400, 'day': 86400}
        return n * mult.get(unit, 60)
    return 3600  # 預設 1 小時


class CronJobStore:

    def __init__(self) -> None:
        self._path = get_agent_home() / "cron_jobs.json"

    def _load(self) -> dict[str, dict]:
        if not self._path.exists():
            return {}
        try:
            return json.loads(self._path.read_text())
        except:
            return {}

    def _save(self, jobs: dict[str, dict]) -> None:
        self._path.write_text(json.dumps(jobs, indent=2))

    def create(
        self,
        name: str,
        schedule: str,
        prompt: str = "",
        no_agent: bool = False,
        script: str = "",
        toolsets: list[str] | None = None,
        workdir: str = "",
    ) -> CronJob:
        now = time.time()
        secs = _parse_schedule(schedule)
        job = CronJob(
            id=uuid.uuid4().hex[:8],
            name=name,
            schedule=schedule,
            schedule_seconds=secs,
            prompt=prompt,
            no_agent=no_agent,
            script=script,
            enabled=True,
            last_run=0,
            next_run=now + secs,
            created_at=now,
            toolsets=toolsets or ["web", "terminal"],
            workdir=workdir,
        )
        jobs = self._load()
        jobs[job.id] = asdict(job)
        self._save(jobs)
        return job

    def due_jobs(self) -> list[CronJob]:
        """回傳所有已到期且啟用的 job。"""
        now = time.time()
        jobs = self._load()
        due = []
        for d in jobs.values():
            if d.get("enabled") and d.get("next_run", 0) <= now:
                due.append(CronJob(**d))
        return due

    def mark_ran(self, job_id: str) -> None:
        """更新 last_run 和 next_run。"""
        jobs = self._load()
        if job_id in jobs:
            now = time.time()
            jobs[job_id]["last_run"] = now
            jobs[job_id]["next_run"] = now + jobs[job_id]["schedule_seconds"]
            jobs[job_id]["run_count"] = jobs[job_id].get("run_count", 0) + 1
            self._save(jobs)

    def list_all(self) -> list[CronJob]:
        return [CronJob(**d) for d in self._load().values()]

    def toggle(self, job_id: str, enabled: bool) -> bool:
        jobs = self._load()
        if job_id in jobs:
            jobs[job_id]["enabled"] = enabled
            self._save(jobs)
            return True
        return False

    def delete(self, job_id: str) -> bool:
        jobs = self._load()
        if job_id in jobs:
            del jobs[job_id]
            self._save(jobs)
            return True
        return False

Scheduler Tick Loop

pythoncron/scheduler.py
"""cron/scheduler.py — Cron Tick Loop"""
import asyncio, logging, time
from cron.job_store import CronJobStore, CronJob

logger = logging.getLogger(__name__)


async def _run_no_agent_job(job: CronJob) -> str:
    """執行 no-agent 腳本，回傳 stdout 輸出。"""
    import io, contextlib
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            exec(job.script, {"__name__": "__cron__"})
        return buf.getvalue()
    except Exception as e:
        return f"[Script error] {type(e).__name__}: {e}"


async def _run_agent_job(job: CronJob) -> str:
    """建立 AIAgent 執行 job prompt，回傳 final_response。"""
    from agent.core import AIAgent
    from agent.providers.resolver import ProviderConfig
    from tools.registry import registry
    from tools.toolsets import resolve_toolset_names
    from pathlib import Path

    # Cron session 使用自動偵測 provider
    cfg = ProviderConfig(provider="auto", model="claude-haiku-4")

    # 把 toolset 名稱解析成 schema 和 handler
    tool_names = resolve_toolset_names(job.toolsets, platform="batch")
    tool_schemas = registry.get_schemas(toolsets=job.toolsets)

    # 注意：cron session 不允許建立新的 cron job（防止遞迴排程）
    tool_schemas = [s for s in tool_schemas if s.get("name") != "cronjob"]

    agent = AIAgent(
        provider_cfg=cfg,
        system_prompt=f"You are an automated agent. Execute the following task precisely: {job.name}",
        max_iterations=15,
        tool_schemas=tool_schemas,
        cwd=Path(job.workdir) if job.workdir else None,
    )
    result = agent.run_conversation(job.prompt)
    return result.final_response


async def run_scheduler(tick_interval: int = 60) -> None:
    """主 tick loop，每 tick_interval 秒喚醒一次。"""
    store = CronJobStore()
    logger.info("Cron scheduler started")

    while True:
        try:
            due = store.due_jobs()
            for job in due:
                logger.info(f"Running cron job: {job.name} ({job.id})")
                try:
                    if job.no_agent:
                        result = await _run_no_agent_job(job)
                    else:
                        result = await _run_agent_job(job)
                    logger.info(f"Job '{job.name}' completed: {result[:100]}")
                except Exception as e:
                    logger.error(f"Job '{job.name}' failed: {e}")
                finally:
                    store.mark_ran(job.id)
        except Exception as e:
            logger.error(f"Scheduler tick error: {e}")

        await asyncio.sleep(tick_interval)

Subagent Delegation：並行任務分發

delegate_task 建立子 AIAgent 實例，每個子代理有完全隔離的 context， 並行處理各自的任務，只把最終摘要回傳給父代理。

pythontools/delegation_tool.py（完整實作）
"""
tools/delegation_tool.py — Subagent Delegation Tool
建立子 AIAgent 實例，並行執行獨立任務。
"""
import asyncio, json
from tools.registry import registry

DELEGATE_SCHEMA = {
    "name": "delegate_task",
    "description": (
        "Spawn one or more subagents to handle independent tasks in parallel. "
        "Each subagent has isolated context. Only final summaries are returned."
    ),
    "parameters": {
        "type": "object",
        "properties": {
            "tasks": {
                "type": "array",
                "description": "List of tasks for subagents",
                "items": {
                    "type": "object",
                    "properties": {
                        "goal": {"type": "string", "description": "Task goal"},
                        "context": {"type": "string", "description": "All info the subagent needs"},
                        "toolsets": {"type": "array", "items": {"type": "string"}},
                    },
                    "required": ["goal"],
                },
            },
            "max_concurrent": {
                "type": "integer",
                "description": "Max parallel subagents (default 3)",
                "default": 3,
            },
        },
        "required": ["tasks"],
    },
}


async def _run_subagent(goal: str, context: str, toolsets: list[str]) -> dict:
    """
    建立一個完全隔離的子 AIAgent 並執行任務。
    子代理看不到父代理的任何對話歷史。
    """
    from agent.core import AIAgent
    from agent.providers.resolver import ProviderConfig
    from tools.registry import registry

    cfg = ProviderConfig(provider="auto", model="claude-haiku-4")
    schemas = registry.get_schemas(toolsets=toolsets or ["web", "file"])

    # 子代理的 system prompt 只包含目標，不包含父代理的記憶或 skills
    agent = AIAgent(
        provider_cfg=cfg,
        system_prompt=f"You are a focused subagent. Complete the following task precisely.\nTask: {goal}",
        max_iterations=15,
        tool_schemas=schemas,
    )

    try:
        result = agent.run_conversation(context if context else f"Please complete: {goal}")
        return {
            "goal": goal,
            "success": not result.stopped_by_budget,
            "result": result.final_response,
            "iterations": result.iterations_used,
        }
    except Exception as e:
        return {"goal": goal, "success": False, "result": f"Subagent error: {e}", "iterations": 0}


def _delegation_handler(args: dict) -> str:
    tasks = args.get("tasks", [])
    max_concurrent = args.get("max_concurrent", 3)
    if not tasks:
        return json.dumps({"error": "tasks list is empty"})

    async def _run_all() -> list:
        sem = asyncio.Semaphore(max_concurrent)
        async def _bounded(task):
            async with sem:
                return await _run_subagent(
                    task.get("goal", ""),
                    task.get("context", ""),
                    task.get("toolsets", ["web"]),
                )
        return await asyncio.gather(*[_bounded(t) for t in tasks])

    try:
        loop = asyncio.get_event_loop()
        results = loop.run_until_complete(_run_all())
    except RuntimeError:
        results = asyncio.run(_run_all())

    successes = [r for r in results if r["success"]]
    return json.dumps({
        "total": len(tasks),
        "succeeded": len(successes),
        "results": results,
    }, ensure_ascii=False)


registry.register(name="delegate_task", toolset="delegation", schema=DELEGATE_SCHEMA, handler=_delegation_handler)

為何子代理看不到父代理的歷史

子代理以完全空白的對話開始，對父代理的對話歷史、記憶、tool call 一無所知。 這是刻意的設計，不是限制。

為何要隔離？

父代理的完整 context 可能有 50K+ tokens。把這些全部傳給每個子代理， token 費用會爆炸。隔離讓子代理用最小的 context 完成明確的子任務。

怎麼讓子代理有足夠資訊？

在 context 欄位明確提供所有子代理需要的資訊。 不要假設子代理知道任何東西——把它當作一個從未與你對話的全新 agent。

⚠ 最常見的 Delegation 錯誤
錯誤：把子任務的 context 欄位留空，假設子代理「知道」父代理之前說過什麼。
後果：子代理不知道專案路徑、認證資訊、限制條件，執行出完全不符合預期的結果。
正確做法：在 context 中提供完整的背景資訊——就算看起來很冗長也是正確的。
設計決策日誌 #009

決策：子代理不繼承父代理的 history，也不繼承記憶 snapshot。
理由：子代理是「工具」，不是「分身」。它的目標是高效完成一個明確的子任務， 而不是複製父代理的完整人格和知識。繼承父代理的 20K tokens system prompt 只會增加費用， 不會提升任務完成品質。
例外：若子任務確實需要某些記憶（如環境資訊），明確地把相關內容複製到 context 欄位。

execute_code Tool：多步驟壓縮成單一 Turn

execute_code 讓 agent 撰寫 Python 腳本，在腳本中直接呼叫 tools， 把多個 tool call 壓縮成一個 LLM turn——只有最終 print() 輸出進入 context。

pythontools/execute_code_tool.py（簡化版實作）
"""
tools/execute_code_tool.py — Execute Python code with tool access
把多步驟 tool 呼叫壓縮成單一 LLM turn。

注意：完整版 Hermes 用 Unix socket RPC 讓腳本非同步呼叫 tools。
這是簡化版：直接用 registry.dispatch() 作為 tool 呼叫介面。
"""
import io, json, contextlib
from tools.registry import registry

EXECUTE_CODE_SCHEMA = {
    "name": "execute_code",
    "description": (
        "Execute a Python script that can call tools directly. "
        "Use when a task requires 3+ tool calls or complex looping logic. "
        "Intermediate tool results stay in the script scope, not in context. "
        "Only print() output is returned."
    ),
    "parameters": {
        "type": "object",
        "properties": {
            "script": {
                "type": "string",
                "description": (
                    "Python script. Use tool(name, **kwargs) to call tools. "
                    "Example: result = tool('terminal', command='ls -la')"
                ),
            },
        },
        "required": ["script"],
    },
}


def _execute_code_handler(args: dict) -> str:
    script = args.get("script", "")
    if not script.strip():
        return json.dumps({"error": "script is empty"})

    buf = io.StringIO()
    err_buf = io.StringIO()

    def tool_caller(name: str, **kwargs) -> dict:
        """腳本內呼叫 tool 的介面函數。"""
        result_str = registry.dispatch(name, kwargs)
        try:
            return json.loads(result_str)
        except:
            return {"output": result_str}

    sandbox_globals = {
        "tool": tool_caller,
        "json": json,
        "print": lambda *a, **kw: print(*a, **kw, file=buf),
        "__name__": "__execute_code__",
    }

    try:
        with contextlib.redirect_stderr(err_buf):
            exec(script, sandbox_globals)
        output = buf.getvalue()
        err = err_buf.getvalue()
        return json.dumps({
            "output": output.strip() or "(no output)",
            "stderr": err.strip() if err.strip() else None,
        }, ensure_ascii=False)
    except Exception as e:
        return json.dumps({"error": f"{type(e).__name__}: {e}", "output": buf.getvalue()})


registry.register(name="execute_code", toolset="code_execution", schema=EXECUTE_CODE_SCHEMA, handler=_execute_code_handler)

Persistent Goals（Ralph Loop）

Persistent Goals 讓 agent 設定一個跨 turn 的長期目標， 在每次 session 開始時檢查目標是否達成，未達成則繼續追蹤。

pythonagent/goals.py（Goal 追蹤系統）
"""
agent/goals.py — Persistent Goals
跨 session 的長期目標追蹤，在 ephemeral layer 注入當前目標狀態。
"""
from __future__ import annotations
import json, time
from pathlib import Path
from agent.prompt_builder import get_agent_home


class GoalStore:
    def __init__(self) -> None:
        self._path = get_agent_home() / "goals.json"

    def _load(self) -> dict:
        if not self._path.exists():
            return {}
        try:
            return json.loads(self._path.read_text())
        except:
            return {}

    def set_goal(self, goal_text: str) -> None:
        data = self._load()
        data["current"] = {
            "text": goal_text,
            "set_at": time.time(),
            "check_count": 0,
            "status": "active",
        }
        self._path.write_text(json.dumps(data, indent=2))

    def get_current(self) -> dict | None:
        return self._load().get("current")

    def complete_goal(self, note: str = "") -> None:
        data = self._load()
        if "current" in data:
            data["current"]["status"] = "completed"
            data["current"]["completed_at"] = time.time()
            data["current"]["note"] = note
            data.setdefault("history", []).append(data.pop("current"))
            self._path.write_text(json.dumps(data, indent=2))

    def clear_goal(self) -> None:
        data = self._load()
        data.pop("current", None)
        self._path.write_text(json.dumps(data, indent=2))

    def to_ephemeral_layer(self) -> str:
        """生成注入 ephemeral layer 的目標提示文字。"""
        goal = self.get_current()
        if not goal or goal.get("status") != "active":
            return ""
        return (
            f"PERSISTENT GOAL (still active): {goal['text']}\n"
            "Check if this goal has been achieved. If so, call goal_complete. "
            "If not, continue working towards it."
        )
💡 Goal 整合進 Ephemeral Layer
把 goal_store.to_ephemeral_layer() 的輸出加入 build_ephemeral_layers() 中（Layer 10）， 這樣每次 API call 都會提醒 agent 當前的長期目標，但不影響 Layers 1-8 的 cache 穩定性。
