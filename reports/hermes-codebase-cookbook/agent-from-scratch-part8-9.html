pythongateway/runner.py
"""gateway/runner.py — GatewayRunner：協調多個 platform adapter"""
from __future__ import annotations
import asyncio, logging
from gateway.base_adapter import PlatformAdapter, IncomingMessage
from gateway.session_router import SessionRouter

logger = logging.getLogger(__name__)


class GatewayRunner:
    """
    管理多個 platform adapter，把每條進入訊息路由到對應 session。
    每個 adapter 在獨立的 asyncio task 中運行。
    """

    def __init__(self) -> None:
        self._adapters: list[PlatformAdapter] = []
        self._router = SessionRouter()

    def add_adapter(self, adapter: PlatformAdapter) -> None:
        self._adapters.append(adapter)

    async def _handle_message(self, adapter: PlatformAdapter, msg: IncomingMessage) -> None:
        """處理單條進入訊息：路由到 session，執行 agent，投遞回覆。"""
        if not adapter.is_authorized(msg.user_id):
            await adapter.send_message(msg.chat_id, "Sorry, you are not authorized.")
            return

        session = self._router.get_or_create(
            platform=msg.platform, user_id=msg.user_id, chat_id=msg.chat_id,
        )

        # 用中斷模式決定如何處理正在執行的任務
        if session.is_running:
            handled = await session.handle_interruption(msg)
            if handled:
                return

        try:
            session.is_running = True
            await adapter.send_typing(msg.chat_id)
            result = await asyncio.get_event_loop().run_in_executor(
                None, session.agent.run_conversation, msg.content
            )
            await adapter.send_message(msg.chat_id, result.final_response)
        except Exception as e:
            logger.error(f"Session error: {e}")
            await adapter.send_message(msg.chat_id, f"An error occurred: {e}")
        finally:
            session.is_running = False

    async def run(self) -> None:
        """啟動所有 adapter 並開始監聽訊息。"""
        tasks = []
        for adapter in self._adapters:
            await adapter.start(lambda msg, a=adapter: self._handle_message(a, msg))
            tasks.append(asyncio.create_task(adapter.listen()))
        logger.info(f"Gateway running with {len(self._adapters)} adapter(s)")
        await asyncio.gather(*tasks)

PlatformAdapter ABC

pythongateway/base_adapter.py
"""gateway/base_adapter.py — Platform Adapter 抽象基類"""
from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Callable, Awaitable
import os


@dataclass
class IncomingMessage:
    """platform adapter 傳給 GatewayRunner 的統一格式。"""
    platform: str
    user_id: str
    chat_id: str
    content: str
    raw: dict = field(default_factory=dict)
    is_voice: bool = False


class PlatformAdapter(ABC):
    """
    所有 platform adapter 的抽象基類。
    每個平台（Telegram / Discord / Slack / Webhook）
    實作這三個抽象方法。
    """

    name: str = "base"
    required_env_vars: list[str] = []

    def check_env(self) -> list[str]:
        """回傳缺少的環境變數名稱列表。"""
        return [v for v in self.required_env_vars if not os.environ.get(v)]

    def is_authorized(self, user_id: str) -> bool:
        """預設實作：若 allowed_users 為空則允許所有人。"""
        allowed = self._allowed_users()
        return not allowed or user_id in allowed

    def _allowed_users(self) -> set[str]:
        """從環境變數讀取白名單。子類別可覆蓋。"""
        raw = os.environ.get("GATEWAY_ALLOWED_USERS", "")
        return set(raw.split(",")) if raw.strip() else set()

    @abstractmethod
    async def start(self, message_callback: Callable[[IncomingMessage], Awaitable]) -> None:
        """初始化 SDK 連線，設定 message_callback。"""

    @abstractmethod
    async def send_message(self, chat_id: str, text: str) -> None:
        """投遞回覆到平台。"""

    async def send_typing(self, chat_id: str) -> None:
        """傳送「正在輸入」提示（選配，預設無動作）。"""

    @abstractmethod
    async def listen(self) -> None:
        """持續監聽訊息的 loop（長時間運行）。"""

Session Routing：三元組 → 獨立 Session

pythongateway/session_router.py
"""
gateway/session_router.py
把 (platform, user_id, chat_id) 映射到獨立的 AIAgent session。
"""
from __future__ import annotations
import asyncio
from dataclasses import dataclass, field
from agent.core import AIAgent
from agent.providers.resolver import ProviderConfig
from agent.session_db import SessionDB


@dataclass
class GatewaySession:
    """一個活躍的 Gateway session 狀態。"""
    agent: AIAgent
    session_id: str
    platform: str
    user_id: str
    chat_id: str
    is_running: bool = False
    interrupt_mode: str = "interrupt"  # interrupt | queue | steer
    _queue: list = field(default_factory=list)

    async def handle_interruption(self, msg) -> bool:
        """
        處理 agent 正在執行時收到的新訊息。
        回傳 True 表示此訊息已被處理（不再走正常流程）。
        """
        if self.interrupt_mode == "queue":
            self._queue.append(msg)
            return True   # 排隊，等當前任務完成後處理
        elif self.interrupt_mode == "steer":
            # 把新訊息附加到 agent 的 history，讓它改變方向
            self.agent.history.append({"role": "user", "content": msg.content})
            return True
        else:  # interrupt（預設）
            return False  # 讓 GatewayRunner 的 executor 自然完成後重新處理


class SessionRouter:

    def __init__(self) -> None:
        self._sessions: dict[tuple, GatewaySession] = {}
        self._db = SessionDB()

    def get_or_create(
        self,
        platform: str,
        user_id: str,
        chat_id: str,
    ) -> GatewaySession:
        """
        Session 的鍵是 (platform, user_id, chat_id) 三元組。
        同一個 Telegram 用戶在不同群組裡對話是兩個獨立的 session。
        """
        key = (platform, user_id, chat_id)
        if key not in self._sessions:
            cfg = ProviderConfig(provider="auto", model="claude-sonnet-4")
            agent = AIAgent(
                provider_cfg=cfg,
                platform=platform,
            )
            sid = self._db.new_session(source=platform, user_id=user_id)
            self._sessions[key] = GatewaySession(
                agent=agent, session_id=sid,
                platform=platform, user_id=user_id, chat_id=chat_id,
            )
        return self._sessions[key]

    def close(self, platform: str, user_id: str, chat_id: str) -> None:
        key = (platform, user_id, chat_id)
        self._sessions.pop(key, None)

使用者授權：白名單 + DM Pairing

授權方式環境變數適用場景
全域白名單GATEWAY_ALLOWED_USERS=123456,789012固定的一組授權用戶
平台白名單TELEGRAM_ALLOWED_USERS=123456只對特定平台設白名單
完全開放GATEWAY_ALLOW_ALL_USERS=true內部測試、私有 bot
DM Pairing無（一次性配對碼）允許新用戶自助授權
pythongateway/auth.py — DM Pairing 實作
"""gateway/auth.py — DM Pairing 一次性配對碼授權"""
import json, secrets, time
from pathlib import Path
from agent.prompt_builder import get_agent_home

PAIRING_CODE_TTL = 600  # 10 分鐘有效


class PairingManager:
    def __init__(self) -> None:
        self._path = get_agent_home() / "pairing_codes.json"
        self._allowed_path = get_agent_home() / "authorized_users.json"

    def _load(self) -> dict:
        if not self._path.exists(): return {}
        try: return json.loads(self._path.read_text())
        except: return {}

    def generate_code(self) -> str:
        """生成一個 10 分鐘有效的配對碼（8 位數字）。"""
        code = "".join([str(secrets.randbelow(10)) for _ in range(8)])
        codes = self._load()
        codes[code] = time.time() + PAIRING_CODE_TTL
        self._path.write_text(json.dumps(codes))
        return code

    def approve(self, code: str, platform: str, user_id: str) -> bool:
        """
        用配對碼授權一個使用者。
        成功回傳 True，code 使用後立即作廢。
        """
        codes = self._load()
        expiry = codes.get(code, 0)
        if time.time() > expiry:
            return False  # 過期或不存在

        # 把用戶加入授權名單
        allowed = {}
        if self._allowed_path.exists():
            try: allowed = json.loads(self._allowed_path.read_text())
            except: allowed = {}
        allowed.setdefault(platform, [])
        if user_id not in allowed[platform]:
            allowed[platform].append(user_id)
        self._allowed_path.write_text(json.dumps(allowed, indent=2))

        # 作廢 code
        del codes[code]
        self._path.write_text(json.dumps(codes))
        return True

    def is_authorized(self, platform: str, user_id: str) -> bool:
        if not self._allowed_path.exists(): return False
        try:
            allowed = json.loads(self._allowed_path.read_text())
            return user_id in allowed.get(platform, [])
        except: return False

中斷模式：三種策略

模式行為適用場景
interrupt（預設）新訊息等 agent 當前 API call 完成後再處理；若 agent 還在跑 tool，等工具結束後中斷一般使用
queue新訊息排隊，等當前任務100%完成後依序處理批次處理、不想被打斷的長任務
steer把新訊息附加到 agent history，讓 agent 在當前任務中途改變方向複雜的互動式任務
ℹ Hermes 的 SIGTERM 中斷機制
Hermes 完整版在 interrupt 模式下，對正在執行 terminal 命令的子行程傳送 SIGTERM（1 秒後 SIGKILL）， 確保 shell 命令真正被終止。我們的簡化版讓當前的 API call 自然完成後中斷。 Part 12 對照 Hermes 原始碼時會深度解析這個 interruptible API call 機制。

"""gateway/platforms/telegram_adapter.py — Telegram platform adapter"""
from __future__ import annotations
import logging, os
from telegram import Update
from telegram.ext import Application, ContextTypes, MessageHandler, filters
from gateway.base_adapter import PlatformAdapter, IncomingMessage

logger = logging.getLogger(__name__)


class TelegramAdapter(PlatformAdapter):

    name = "telegram"
    required_env_vars = ["TELEGRAM_BOT_TOKEN"]

    def __init__(self) -> None:
        self._token = os.environ.get("TELEGRAM_BOT_TOKEN", "")
        self._allowed = set(
            os.environ.get("TELEGRAM_ALLOWED_USERS", "").split(",")
        ) - {""}
        self._app: Application | None = None
        self._callback = None

    def _allowed_users(self) -> set[str]:
        # Telegram 優先使用 TELEGRAM_ALLOWED_USERS，fallback 到全域設定
        return self._allowed or super()._allowed_users()

    async def start(self, message_callback) -> None:
        self._callback = message_callback
        self._app = Application.builder().token(self._token).build()

        async def on_message(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
            if not update.message or not update.message.text:
                return
            msg = IncomingMessage(
                platform="telegram",
                user_id=str(update.message.from_user.id),
                chat_id=str(update.message.chat_id),
                content=update.message.text,
                raw=update.to_dict(),
            )
            await message_callback(msg)

        self._app.add_handler(MessageHandler(filters.TEXT, on_message))
        await self._app.initialize()
        await self._app.start()
        logger.info("Telegram adapter initialized")

    async def send_message(self, chat_id: str, text: str) -> None:
        if not self._app:
            return
        # Telegram 單訊息上限 4096 chars；自動分割長訊息
        for chunk in _split_message(text, 4000):
            await self._app.bot.send_message(chat_id=int(chat_id), text=chunk)

    async def send_typing(self, chat_id: str) -> None:
        if self._app:
            await self._app.bot.send_chat_action(chat_id=int(chat_id), action="typing")

    async def listen(self) -> None:
        if self._app:
            await self._app.updater.start_polling()
            await self._app.updater.idle()


def _split_message(text: str, max_len: int) -> list[str]:
    if len(text) <= max_len:
        return [text]
    chunks = []
    while text:
        chunk = text[:max_len]
        # 盡量在換行或句號切割
        cut = chunk.rfind("\n") or chunk.rfind("。") or chunk.rfind(".")
        if cut > max_len * 0.7:
            chunk = text[:cut + 1]
        chunks.append(chunk)
        text = text[len(chunk):].lstrip()
    return chunks

Webhook 事件驅動：GitHub PR → Agent

pythongateway/platforms/webhook_adapter.py（簡化版）
"""
gateway/platforms/webhook_adapter.py
接收 HTTP webhook 事件，轉換成 agent 任務。
範例：GitHub PR opened → agent 自動進行 code review。
"""
from aiohttp import web
from gateway.base_adapter import PlatformAdapter, IncomingMessage


class WebhookAdapter(PlatformAdapter):

    name = "webhook"
    required_env_vars = ["WEBHOOK_SECRET"]

    def __init__(self, host: str = "0.0.0.0", port: int = 8080) -> None:
        self.host = host
        self.port = port
        self._callback = None
        self._app = web.Application()

    async def start(self, message_callback) -> None:
        self._callback = message_callback
        self._app.router.add_post("/webhook/github", self._github_handler)
        self._app.router.add_post("/webhook/generic", self._generic_handler)

    async def _github_handler(self, request: web.Request) -> web.Response:
        payload = await request.json()
        event = request.headers.get("X-GitHub-Event", "")

        if event == "pull_request" and payload.get("action") == "opened":
            pr = payload["pull_request"]
            msg = IncomingMessage(
                platform="webhook",
                user_id="github-bot",
                chat_id=f"pr-{pr['number']}",
                content=(
                    f"New PR opened: {pr['title']}\n"
                    f"URL: {pr['html_url']}\n"
                    f"Please do a code review and post a summary as a PR comment."
                ),
                raw=payload,
            )
            if self._callback:
                await self._callback(msg)

        return web.Response(text="ok")

    async def _generic_handler(self, request: web.Request) -> web.Response:
        payload = await request.json()
        msg = IncomingMessage(
            platform="webhook",
            user_id=payload.get("user_id", "webhook"),
            chat_id=payload.get("chat_id", "default"),
            content=payload.get("message", ""),
            raw=payload,
        )
        if self._callback:
            await self._callback(msg)
        return web.Response(text="ok")

    async def send_message(self, chat_id: str, text: str) -> None:
        # Webhook adapter 的回覆通常由 agent 的 tool 呼叫投遞（如 gh pr comment）
        print(f"[Webhook reply to {chat_id}]: {text[:200]}")

    async def listen(self) -> None:
        runner = web.AppRunner(self._app)
        await runner.setup()
        site = web.TCPSite(runner, self.host, self.port)
        await site.start()
        print(f"Webhook listening on http://{self.host}:{self.port}")
        await asyncio.sleep(3600  * 24 * 365)  # 持續運行

Part 9 · Plugin 系統

設計哲學：Drop-in Directory

為何不用 pip entry_points？

Entry points 需要 pip install，對個人用途過於複雜。Drop-in directory 只要把 plugin 目錄放進去，下次啟動自動載入——零部署步驟。

三個發現來源

1. ~/.myagent/plugins/（全域個人）
2. .myagent/plugins/（專案本地，opt-in）
3. pip entry_points（發布的套件）

Plugin 的最大能力

新增 tool、攔截 tool call（pre/post hook）、替換記憶 provider（single-select）、替換 context engine、新增 LLM provider、覆蓋內建 tool。

Plugin 不能做的事

不能修改 agent 的 conversation_history（保持隔離）。不能覆蓋安全相關的 approval flow（只能在 post_tool_call 觀察）。

"""plugins/loader.py — 掃描三個來源並載入所有 plugin"""
from __future__ import annotations
import importlib, importlib.util, logging, os, sys
from pathlib import Path
import yaml
from agent.prompt_builder import get_agent_home

logger = logging.getLogger(__name__)


def discover_plugin_dirs() -> list[Path]:
    """
    按優先序列出所有 plugin 目錄：
    1. ~/.myagent/plugins/（全域）
    2. .myagent/plugins/（專案本地，若 MYAGENT_ENABLE_PROJECT_PLUGINS=true）
    """
    dirs = []
    global_plugins = get_agent_home() / "plugins"
    if global_plugins.exists():
        dirs.append(global_plugins)

    if os.environ.get("MYAGENT_ENABLE_PROJECT_PLUGINS", "false").lower() == "true":
        local_plugins = Path.cwd() / ".myagent" / "plugins"
        if local_plugins.exists():
            dirs.append(local_plugins)
    return dirs


def _load_plugin_manifest(plugin_dir: Path) -> dict | None:
    yaml_path = plugin_dir / "plugin.yaml"
    if not yaml_path.exists():
        return None
    try:
        m = yaml.safe_load(yaml_path.read_text()) or {}
        return m if m.get("enabled", True) else None
    except:
        return None


def load_all_plugins(ctx) -> int:
    """
    載入所有已發現的 plugin。
    ctx 是 PluginContext 實例。
    回傳成功載入的 plugin 數量。
    """
    loaded = 0
    for plugins_dir in discover_plugin_dirs():
        for plugin_dir in sorted(plugins_dir.iterdir()):
            if not plugin_dir.is_dir(): continue
            manifest = _load_plugin_manifest(plugin_dir)
            if manifest is None: continue

            init_file = plugin_dir / "__init__.py"
            if not init_file.exists(): continue

            try:
                spec = importlib.util.spec_from_file_location(
                    f"plugins.{plugin_dir.name}", init_file
                )
                mod = importlib.util.module_from_spec(spec)
                sys.modules[spec.name] = mod
                spec.loader.exec_module(mod)
                if hasattr(mod, "register"):
                    mod.register(ctx)
                    loaded += 1
                    logger.info(f"Loaded plugin: {manifest.get('name', plugin_dir.name)}")
            except Exception as e:
                logger.error(f"Failed to load plugin '{plugin_dir.name}': {e}")
    return loaded

"""plugins/context.py — PluginContext：plugin 的唯一操作介面"""
from __future__ import annotations
from typing import Any, Callable
from tools.registry import registry as _registry


class PluginContext:
    """
    Plugin 的操作介面。
    每個 plugin 的 register(ctx) 函數接收這個物件。
    """

    def __init__(self) -> None:
        self._hooks: dict[str, list[Callable]] = {}
        self._slash_commands: dict[str, dict] = {}
        self._memory_provider: Callable | None = None
        self._context_engine: Any | None = None
        self._tool_overrides: dict[str, Callable] = {}

    # ── Tool 管理 ──────────────────────────────────────────────

    def register_tool(
        self,
        name: str,
        schema: dict,
        handler: Callable,
        toolset: str = "plugin",
    ) -> None:
        """新增自訂 tool 到 global registry。"""
        _registry.register(name=name, schema=schema, handler=handler, toolset=toolset)

    def tool_override(self, name: str, handler: Callable) -> None:
        """
        替換內建 tool 的 handler。
        注意：schema 保持不變，只有執行邏輯被替換。
        """
        self._tool_overrides[name] = handler
        if name in _registry._tools:
            _registry._tools[name].handler = handler

    # ── Lifecycle Hooks ────────────────────────────────────────

    def register_hook(self, event: str, handler: Callable) -> None:
        """
        訂閱 lifecycle event。
        可用的 event：
          pre_tool_call, post_tool_call,
          agent:start, agent:end, agent:error,
          memory:flush, gateway:message_received
        """
        self._hooks.setdefault(event, []).append(handler)
        # 同步到 tool registry（tool-level hooks）
        if event in ("pre_tool_call", "post_tool_call"):
            _registry.add_hook(event, handler)

    def dispatch_hook(self, event: str, context: dict) -> None:
        """觸發指定 event 的所有 hook handler。"""
        for handler in self._hooks.get(event, []):
            try:
                handler(context)
            except Exception as e:
                import logging
                logging.getLogger(__name__).warning(f"Hook '{event}' error: {e}")

    # ── Slash Commands ─────────────────────────────────────────

    def register_slash_command(
        self,
        name: str,
        handler: Callable,
        help: str = "",
    ) -> None:
        """新增自訂 slash 命令（如 /mystatus）。"""
        self._slash_commands[name] = {"handler": handler, "help": help}

    # ── Provider Overrides ─────────────────────────────────────

    def register_memory_provider(self, provider: Callable) -> None:
        """
        替換記憶 provider（single-select：只能有一個活躍的）。
        provider 必須實作 read() / write() / search() 介面。
        """
        self._memory_provider = provider

    def register_context_engine(self, engine) -> None:
        """替換 context 壓縮引擎（single-select）。"""
        self._context_engine = engine

Memory Provider Plugin（Single-Select）

預設的記憶 provider 是本地 MEMORY.md 檔案。 Plugin 可以替換成 Honcho、Mem0 或任何遠端服務——但只能有一個活躍的。

python範例：Mem0 Memory Provider Plugin
# ~/.myagent/plugins/mem0-provider/__init__.py
def register(ctx):
    from mem0 import MemoryClient
    import os

    client = MemoryClient(api_key=os.environ["MEM0_API_KEY"])

    class Mem0Provider:
        def read(self) -> str:
            memories = client.get_all(user_id="default")
            return "\n§\n".join(m["memory"] for m in memories)

        def write(self, content: str) -> None:
            client.add(content, user_id="default")

        def search(self, query: str) -> list[str]:
            results = client.search(query, user_id="default")
            return [r["memory"] for r in results]

    ctx.register_memory_provider(Mem0Provider())

from abc import ABC, abstractmethod
from typing import Any

Message = dict[str, Any]

class ContextEngine(ABC):
    """Context 壓縮引擎的抽象基類。"""

    @abstractmethod
    def compress(
        self,
        history: list[Message],
        target_tokens: int,
    ) -> list[Message]:
        """
        把 history 壓縮到 target_tokens 以內。
        回傳壓縮後的 message list（格式與輸入相同）。
        """

    @abstractmethod
    def estimate_tokens(self, messages: list[Message]) -> int:
        """估算 message list 的 token 數量。"""

完整範例：Prometheus Metrics Plugin

這個 plugin 記錄每次 tool call 的延遲和成功率，暴露給 Prometheus scrape。

python~/.myagent/plugins/prometheus-metrics/__init__.py
"""
Prometheus Metrics Plugin
記錄 tool call 延遲、成功率，透過 HTTP /metrics 暴露給 Prometheus。

需要：pip install prometheus_client
"""
import time
from prometheus_client import (
    Counter, Histogram, Gauge, start_http_server
)

# ── Metrics 定義 ───────────────────────────────────────────────
TOOL_CALLS = Counter(
    "agent_tool_calls_total",
    "Total number of tool calls",
    ["tool_name", "status"]
)
TOOL_LATENCY = Histogram(
    "agent_tool_call_duration_seconds",
    "Tool call latency in seconds",
    ["tool_name"],
    buckets=[0.1, 0.5, 1, 5, 30, 60, 120]
)
ACTIVE_SESSIONS = Gauge(
    "agent_active_sessions",
    "Number of currently active sessions"
)
LLM_CALLS = Counter(
    "agent_llm_calls_total",
    "Total LLM API calls",
    ["provider", "status"]
)

# 追蹤 pre_tool_call 的開始時間
_start_times: dict[str, float] = {}


def register(ctx):
    """Plugin 入口點。"""

    def on_pre_tool(context: dict) -> None:
        key = f"{context.get('task_id', 'unknown')}:{context.get('tool_name')}"
        _start_times[key] = time.time()

    def on_post_tool(context: dict) -> None:
        tool_name = context.get("tool_name", "unknown")
        key = f"{context.get('task_id', 'unknown')}:{tool_name}"
        start = _start_times.pop(key, time.time())
        elapsed = time.time() - start

        result = context.get("result", "")
        status = "error" if '"error"' in result[:100] else "success"

        TOOL_CALLS.labels(tool_name=tool_name, status=status).inc()
        TOOL_LATENCY.labels(tool_name=tool_name).observe(elapsed)

    def on_agent_start(context: dict) -> None:
        ACTIVE_SESSIONS.inc()

    def on_agent_end(context: dict) -> None:
        ACTIVE_SESSIONS.dec()

    ctx.register_hook("pre_tool_call", on_pre_tool)
    ctx.register_hook("post_tool_call", on_post_tool)
    ctx.register_hook("agent:start", on_agent_start)
    ctx.register_hook("agent:end", on_agent_end)

    # 啟動 Prometheus HTTP server（port 9100）
    import threading
    threading.Thread(
        target=lambda: start_http_server(9100), daemon=True
    ).start()
    print("[prometheus-metrics] Metrics available at http://localhost:9100/metrics")
