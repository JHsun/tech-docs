"""
run_agent.py — 完整 Agent 入口點
整合所有 Part 1-10 的子系統。
"""
import asyncio, logging, os
from pathlib import Path
from dotenv import load_dotenv

from agent.core import AIAgent
from agent.providers.resolver import ProviderConfig
from agent.session_db import SessionDB
from tools.registry import registry
from plugins.loader import load_all_plugins
from plugins.context import PluginContext

logging.basicConfig(level=logging.INFO, format="%(levelname)s %(name)s: %(message)s")
logger = logging.getLogger(__name__)


def build_agent(
    toolsets=None,
    platform: str = "cli",
    cwd: Path | None = None,
) -> AIAgent:
    """完整組裝 AIAgent：provider + tools + plugins。"""
    load_dotenv()

    # 1. Provider（自動偵測）
    cfg = ProviderConfig(provider="auto", model="claude-sonnet-4")

    # 2. Tool Registry — 自動發現所有 tools/ 模組
    registry.discover("tools")
    logger.info(f"Discovered {len(registry._tools)} tools")

    # 3. Plugin System — 載入所有 plugins
    plugin_ctx = PluginContext()
    n_plugins = load_all_plugins(plugin_ctx)
    logger.info(f"Loaded {n_plugins} plugins")

    # 4. Tool Schemas（根據 toolsets 過濾）
    from tools.toolsets import resolve_toolset_names
    active_toolsets = toolsets or ["terminal", "file", "web", "memory", "skills"]
    schemas = registry.get_schemas(toolsets=active_toolsets)

    # 5. 建立 Agent
    agent = AIAgent(
        provider_cfg=cfg,
        max_iterations=25,
        tool_schemas=schemas,
        platform=platform,
        cwd=cwd or Path.cwd(),
    )
    return agent


def run_cli() -> None:
    """互動式 CLI 模式。"""
    agent = build_agent()
    db = SessionDB()
    sid = db.new_session(source="cli")
    logger.info(f"Session: {sid}")

    print("Agent ready. Type 'exit' to quit.\n")
    while True:
        try:
            user_input = input("You: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye.")
            break

        if not user_input or user_input.lower() in ("exit", "quit"):
            break

        result = agent.run_conversation(user_input)
        print(f"\nAgent: {result.final_response}\n")

        # 持久化到 session store
        for msg in result.history[-2:]:  # 最後的 user + assistant
            db.append_message(sid, msg)

    db.end_session(sid)


async def run_gateway_mode() -> None:
    """Gateway 模式（多平台）。"""
    from gateway.runner import GatewayRunner
    from gateway.platforms.telegram_adapter import TelegramAdapter

    load_dotenv()
    registry.discover("tools")

    gw = GatewayRunner()
    if os.environ.get("TELEGRAM_BOT_TOKEN"):
        gw.add_adapter(TelegramAdapter())

    logger.info("Starting gateway...")
    await gw.run()


if __name__ == "__main__":
    import sys
    mode = sys.argv[1] if len(sys.argv) > 1 else "cli"
    if mode == "gateway":
        asyncio.run(run_gateway_mode())
    elif mode == "cron":
        from cron.scheduler import run_scheduler
        asyncio.run(run_scheduler())
    else:
        run_cli()

"""agent/config.py — config.yaml 讀取，env var 覆蓋"""
import os, yaml
from functools import lru_cache
from agent.prompt_builder import get_agent_home


@lru_cache(maxsize=None)
def load_config() -> dict:
    """
    讀取 config.yaml，合併環境變數覆蓋。
    結果被快取——只讀一次，整個程序期間不變。
    """
    config_path = get_agent_home() / "config.yaml"
    cfg: dict = {}
    if config_path.exists():
        try:
            cfg = yaml.safe_load(config_path.read_text()) or {}
        except yaml.YAMLError:
            cfg = {}

    # 環境變數優先覆蓋
    env_overrides = {
        "MYAGENT_MODEL": ("model",),
        "MYAGENT_PROVIDER": ("provider",),
        "MYAGENT_APPROVAL_MODE": ("approvals", "mode"),
    }
    for env_var, path in env_overrides.items():
        val = os.environ.get(env_var)
        if val:
            d = cfg
            for key in path[:-1]:
                d = d.setdefault(key, {})
            d[path[-1]] = val

    return cfg


def get(key: str, default=None):
    """用點號路徑讀取 config 值，如 get('approvals.mode', 'manual')"""
    parts = key.split(".")
    d = load_config()
    for p in parts:
        if not isinstance(d, dict): return default
        d = d.get(p, default)
    return d

"""cli/doctor.py — 系統健康診斷"""
import os
from agent.prompt_builder import get_agent_home


def run_doctor() -> bool:
    """回傳 True 表示所有檢查通過。"""
    checks = []

    def check(name: str, ok: bool, hint: str = "") -> None:
        status = "✅" if ok else "❌"
        msg = f"{status} {name}"
        if not ok and hint:
            msg += f"\n   hint: {hint}"
        print(msg)
        checks.append(ok)

    home = get_agent_home()
    print(f"Agent home: {home}\n")

    # ── 目錄結構 ───────────────────────────────────────
    check("~/.myagent/ exists", home.exists(),
          f"mkdir -p {home}")
    check("~/.myagent/memories/ exists", (home / "memories").exists())
    check("~/.myagent/skills/ exists", (home / "skills").exists())

    # ── API Keys ───────────────────────────────────────
    has_any_key = any([
        os.environ.get("ANTHROPIC_API_KEY"),
        os.environ.get("OPENROUTER_API_KEY"),
        os.environ.get("OPENAI_API_KEY"),
    ])
    check("LLM API key configured", has_any_key,
          "Set ANTHROPIC_API_KEY or OPENROUTER_API_KEY in .env")

    # ── LLM 連線測試 ─────────────────────────────────
    try:
        from agent.providers.resolver import ProviderConfig, resolve_transport
        cfg = ProviderConfig(provider="auto", model="claude-haiku-4")
        transport = resolve_transport(cfg)
        resp = transport.call(
            [{"role": "system", "content": "test"},
             {"role": "user", "content": "reply with: OK"}],
            [], cfg.model
        )
        check("LLM connection", bool(resp.content))
    except Exception as e:
        check("LLM connection", False, str(e))

    # ── Database ──────────────────────────────────────
    try:
        from agent.session_db import SessionDB
        db = SessionDB()
        db.list_sessions(limit=1)
        check("SQLite session DB", True)
    except Exception as e:
        check("SQLite session DB", False, str(e))

    # ── Tool Registry ─────────────────────────────────
    try:
        from tools.registry import registry
        registry.discover("tools")
        n = len(registry._tools)
        check(f"Tool registry ({n} tools)", n > 0)
    except Exception as e:
        check("Tool registry", False, str(e))

    passed = sum(checks)
    print(f"\n{passed}/{len(checks)} checks passed")
    return passed == len(checks)


if __name__ == "__main__":
    run_doctor()

Hermes 的 Interruptible API Call

🔬
深度解析①：在 HTTP request 執行中途中斷 agent

問題：當 Telegram 用戶說「停止」時，agent 可能正在等待一個 30 秒的 LLM API response。 如何做到真正的即時中斷，而不是等 API call 完成？

Hermes 的解法是三執行緒架構：

  1. 主執行緒（Gateway event loop）：監聽訊息，設定 interrupt event
  2. API 執行緒（ThreadPoolExecutor）：執行 LLM API call，輪詢 interrupt event
  3. Terminal 子行程：執行 shell 命令，收到 SIGTERM 後終止
pythonHermes run_agent.py — Interruptible API call 概念實作（簡化版）
import threading
from concurrent.futures import ThreadPoolExecutor


class InterruptibleAgent:
    """
    展示 Hermes 的 interruptible API call 設計。
    完整版在 run_agent.py 的 _interruptible_api_call() 函數。
    """

    def __init__(self) -> None:
        self._interrupt_event = threading.Event()
        self._executor = ThreadPoolExecutor(max_workers=1)
        self._current_future = None

    def interrupt(self) -> None:
        """Gateway 收到中斷訊息時呼叫。"""
        self._interrupt_event.set()
        # 終止正在執行的 terminal 子行程（若有的話）
        self._kill_terminal_children()

    def _call_api(self, messages, schemas, model) -> TransportResponse | None:
        """
        在獨立執行緒中呼叫 LLM API。
        每 0.1 秒檢查 interrupt_event，若被設定則取消請求。
        """
        result: list = []
        error: list = []

        def do_call():
            try:
                r = self._transport.call(messages, schemas, model)
                result.append(r)
            except Exception as e:
                error.append(e)

        t = threading.Thread(target=do_call, daemon=True)
        t.start()

        # 等待完成或中斷信號（100ms 輪詢）
        while t.is_alive():
            if self._interrupt_event.is_set():
                # API call 無法強制終止，但我們可以不使用其結果
                return None  # 表示被中斷
            t.join(timeout=0.1)

        if error:
            raise error[0]
        return result[0] if result else None

    def _kill_terminal_children(self) -> None:
        """
        終止所有由此 agent 建立的 terminal 子行程。
        Hermes 用 os.killpg(os.getpgid(pid), signal.SIGTERM) 終止整個 process group。
        """
        import signal, os
        # Hermes 實作：遍歷所有記錄的子行程 PID，發送 SIGTERM
        # 1 秒後若未結束，發送 SIGKILL
        pass  # 完整實作見 Hermes tools/terminal_tool.py
💡 為什麼這個設計很精妙
  • LLM API call 無法真正被 HTTP 層取消（大多數 SDK 不支援），但我們可以忽略其結果
  • Terminal 子行程可以用 SIGTERM/SIGKILL 真正終止，防止不受控的命令繼續運行
  • interrupt_event 是 threading.Event（非 asyncio.Event），讓它可以從 asyncio 事件迴圈安全地 set
  • 每個 session 有獨立的 interrupt_event，多用戶不會互相影響

Hermes 的 Prompt Cache Breakpoint 最佳化

🔬
深度解析②：動態計算最佳 cache breakpoint 位置

我們的實作把 cache breakpoint 固定放在 Layer 8 末尾。 Hermes 的做法更精確：用 model metadata 估算 token 數， 在不超過 Anthropic cache granularity 上限的前提下，最大化可快取的 prefix 長度。

Anthropic prompt cache 的技術細節：

  • 最小快取單位：1,024 tokens（prefix 至少要這麼長才值得快取）
  • 最多可設定 4 個 cache breakpoint
  • 快取有效期：5 分鐘（每次命中會重置計時器）
  • cache breakpoint 必須設在 stable prefix 的末尾，不能在會改變的部分
pythonagent/prompt_caching.py（Hermes 的策略，簡化說明）
"""
Hermes 的 prompt_caching.py 的核心邏輯（概念說明）：

1. 在 session 開始時，用 model_metadata.estimate_tokens() 估算各層 token 數
2. 從 Layer 1 開始累積 token 計數
3. 在最接近（但不超過）context window 50% 的位置設 breakpoint
4. 如果 MEMORY.md 超過 2,200 chars（大型記憶），
   在 Layer 4 末尾額外設一個 breakpoint（雙重快取）
5. 每次 API call 時動態插入 cache_control marker

實際效果（來自 Hermes 文件）：
- 一般 session：system prompt 命中率 ~95%
- 大型記憶 + 大量 skills：雙 breakpoint 讓命中率維持 ~90%
"""

def apply_cache_breakpoints(
    system_parts: list[dict],   # Anthropic format system list
    stable_token_count: int,     # 估算的穩定 prefix token 數
    model_context_window: int,   # 模型的 context window 大小
) -> list[dict]:
    """
    在 system_parts 的末尾（Layer 8 之後）插入 cache breakpoint。
    若 stable_token_count 超過 context_window 的 40%，
    另在 Layer 4 末尾加第二個 breakpoint。
    """
    if not system_parts:
        return system_parts

    if stable_token_count < 1024:
        return system_parts  # 太短，不值得快取

    # 在最後一個 system part 加 cache breakpoint
    result = list(system_parts)
    last = dict(result[-1])  # 複製一份避免修改原始物件
    last["cache_control"] = {"type": "ephemeral"}
    result[-1] = last

    return result

Part 12 · 深度解析③

Hermes 的 Session Lineage 追蹤

🔬
深度解析③：壓縮後仍能追溯完整對話歷史

Context compression 會丟棄中間的對話。Hermes 用 session lineage 確保這些被壓縮的內容永遠可以被找回。

Hermes 的 session lineage 設計：

  1. 壓縮發生時，把摘要後的對話存為新的 session（有新的 session ID）
  2. 新 session 的 parent_session_id 指向原始 session
  3. 原始 session 完整保存在 SQLite——沒有任何資料被刪除
  4. session_search tool 可以跨越 session 邊界搜尋（FTS5 涵蓋所有歷史）
  5. 用戶執行 /rollback 時，沿著 lineage chain 找到對應的 parent session 恢復
ℹ 為什麼這比「截斷」或「就地壓縮」更好
  • 可審計：任何時候都能還原完整的對話歷史，對 compliance / debugging 至關重要
  • 可 rollback：壓縮是可逆的，agent 可以恢復到壓縮前的任意狀態
  • FTS5 仍可搜尋：被壓縮的 session 的 FTS5 index 仍然存在，/recall 能搜尋到壓縮前的內容
