5.1
Part 5 · 持久記憶

為何用 SQLite

為何不用純 JSONL 檔案？

JSONL 適合 append-only，但搜尋需要全部讀入記憶體再掃描。session 數百條、每條數千 tokens 時，FTS 搜尋會很慢。Hermes 從 JSONL 遷移到 SQLite 的原因之一。

為何不用 Redis？

Redis 需要額外的伺服器程序。SQLite 是嵌入式資料庫——沒有網路連線、沒有額外服務、備份就是複製一個檔案。agent 的 session 存取模式是典型的 single-writer + concurrent-reader，SQLite WAL mode 完全勝任。

FTS5 的優勢

SQLite 的 FTS5 支援全文搜尋，還有 trigram tokenizer 支援 CJK substring 搜尋。Hermes 同時建兩個 FTS5 表（一般 + trigram）以覆蓋英文和中文搜尋場景。

WAL Mode

Write-Ahead Logging 讓一個 writer（Gateway message handler）和多個 reader（CLI、cron、FTS 搜尋）可以同時運行而不互相阻塞。

-- sessions 表：每個對話 session 的 metadata
CREATE TABLE IF NOT EXISTS sessions (
    id            TEXT PRIMARY KEY,  -- 格式：20260518_091523_a1b2c3
    source        TEXT NOT NULL,     -- cli / telegram / discord / cron
    user_id       TEXT,              -- platform user ID
    model         TEXT,              -- 使用的模型
    platform      TEXT,
    started_at    REAL NOT NULL,     -- Unix epoch float
    ended_at      REAL,
    end_reason    TEXT,              -- normal / budget / interrupted
    message_count INTEGER DEFAULT 0,
    input_tokens  INTEGER DEFAULT 0,
    output_tokens INTEGER DEFAULT 0,
    cache_read_tokens INTEGER DEFAULT 0,
    title         TEXT,
    parent_session_id TEXT           -- compression 後的 lineage 追蹤
);

CREATE INDEX IF NOT EXISTS idx_sessions_source  ON sessions(source);
CREATE INDEX IF NOT EXISTS idx_sessions_started ON sessions(started_at DESC);
CREATE UNIQUE INDEX IF NOT EXISTS idx_sessions_title
    ON sessions(title) WHERE title IS NOT NULL;

-- messages 表：每條訊息的完整記錄
CREATE TABLE IF NOT EXISTS messages (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id  TEXT NOT NULL REFERENCES sessions(id),
    role        TEXT NOT NULL,   -- user / assistant / tool
    content     TEXT,
    tool_call_id TEXT,          -- role=tool 時的 ID
    tool_calls  TEXT,           -- JSON string（role=assistant 時）
    tool_name   TEXT,
    timestamp   REAL NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_messages_session
    ON messages(session_id, timestamp);

-- FTS5 全文搜尋（英文）
CREATE VIRTUAL TABLE IF NOT EXISTS messages_fts USING fts5(
    content, content=messages, content_rowid=id
);

-- FTS5 Trigram（中文 / substring 搜尋）
CREATE VIRTUAL TABLE IF NOT EXISTS messages_fts_trigram USING fts5(
    content, content=messages, content_rowid=id,
    tokenize="trigram"
);

FTS5 Triggers 與搜尋實作

FTS5 虛擬表需要手動同步。三個 trigger 保持同步（INSERT / DELETE / UPDATE）：

sqlFTS5 Sync Triggers
CREATE TRIGGER IF NOT EXISTS messages_fts_insert
AFTER INSERT ON messages BEGIN
    INSERT INTO messages_fts(rowid, content) VALUES (new.id, new.content);
    INSERT INTO messages_fts_trigram(rowid, content) VALUES (new.id, new.content);
END;

CREATE TRIGGER IF NOT EXISTS messages_fts_delete
AFTER DELETE ON messages BEGIN
    INSERT INTO messages_fts(messages_fts, rowid, content)
        VALUES('delete', old.id, old.content);
    INSERT INTO messages_fts_trigram(messages_fts_trigram, rowid, content)
        VALUES('delete', old.id, old.content);
END;

搜尋函數：自動偵測 trigram

pythonagent/session_db.py — search_sessions()
def _has_cjk(text: str) -> bool:
    return any('\u4e00' <= c <= '\u9fff' for c in text)


def search_sessions(
    self,
    query: str,
    limit: int = 10,
    source: str | None = None,
) -> list[dict]:
    """
    全文搜尋所有 session 的訊息內容。
    短查詢或含中文 → trigram FTS（支援 substring match）
    否則 → 標準 FTS5
    """
    use_trigram = len(query) < 3 or _has_cjk(query)
    fts = "messages_fts_trigram" if use_trigram else "messages_fts"
    where_source = "AND s.source = ?" if source else ""
    sql = f"""
    SELECT s.id AS session_id, s.title, s.source, s.started_at,
           snippet({fts}, 0, '**', '**', '...', 32) AS snippet, rank
    FROM {fts}
    JOIN messages m ON m.id = {fts}.rowid
    JOIN sessions s ON s.id = m.session_id
    WHERE {fts} MATCH ? {where_source}
    ORDER BY rank LIMIT ?"""
    params = [query] + ([source] if source else []) + [limit]
    with self._conn() as conn:
        return [dict(row) for row in conn.execute(sql, params).fetchall()]

Idempotent Schema Migration

不寫版本化 migration script，改用宣告式欄位對齊——比對目標 schema 和當前 schema 的 diff，自動補上缺少的欄位。天生是 idempotent 的。

pythonagent/session_db.py — _migrate()
def _migrate(self, conn) -> None:
    """
    宣告式欄位對齊（Hermes 的 declarative column reconciliation）：
    1. 先執行 CREATE TABLE IF NOT EXISTS（建立缺少的表）
    2. 對每個表，比對目標欄位 vs 實際欄位，補上缺少的欄位
    
    不能刪除欄位或改欄位型別（SQLite 的限制），但 session store 幾乎不需要。
    """
    conn.executescript(self._SCHEMA_SQL)  # CREATE TABLE / INDEX / TRIGGER / VIRTUAL TABLE

    for table_name in ("sessions", "messages"):
        target_cols = self._parse_schema_columns(table_name)
        actual_cols = {row[1] for row in conn.execute(f"PRAGMA table_info({table_name})").fetchall()}
        for col_name, col_def in target_cols.items():
            if col_name not in actual_cols:
                try:
                    conn.execute(f"ALTER TABLE {table_name} ADD COLUMN {col_name} {col_def}")
                except Exception as e:
                    if "duplicate column" not in str(e).lower():
                        raise

def _parse_schema_columns(self, table_name: str) -> dict[str, str]:
    """從 _SCHEMA_SQL 中解析指定表的欄位定義。"""
    import re
    pattern = rf"CREATE TABLE IF NOT EXISTS {table_name}\s*\(([\s\S]*?)\);"
    match = re.search(pattern, self._SCHEMA_SQL, re.IGNORECASE)
    if not match:
        return {}
    cols = {}
    for line in match.group(1).strip().split("\n"):
        line = line.strip().rstrip(",")
        if not line or line.upper().startswith(("PRIMARY", "FOREIGN", "UNIQUE", "CHECK")):
            continue
        parts = line.split()
        if parts:
            col_name = parts[0]
            col_def = " ".join(parts[1:])
            cols[col_name] = col_def
    return cols
設計決策日誌 #007

決策：不寫版本化 migration script，改用宣告式欄位對齊。
理由：版本化 migration 在分散式開發中常遇到版本衝突。宣告式對齊只問「現在缺哪些欄位」並補上，邏輯更簡單也更健壯。
限制：這個方法不能刪除欄位或改欄位型別，但 agent session store 幾乎不需要這樣做。

完整 SessionDB：CRUD + Resume

pythonagent/session_db.py（完整 SessionDB class）
"""agent/session_db.py — SQLite + FTS5 Session Store"""
from __future__ import annotations
import json, re, sqlite3, time, uuid
from contextlib import contextmanager
from pathlib import Path
from agent.prompt_builder import get_agent_home

# _SCHEMA_SQL 定義（來自 5.2 節的完整 SQL，含 triggers）
# 為節省篇幅此處省略，實際使用時請把 5.2 節的 SQL 貼入此處

class SessionDB:

    def __init__(self, db_path: Path | None = None) -> None:
        self.db_path = db_path or get_agent_home() / "state.db"
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._init_db()

    def _init_db(self) -> None:
        with self._conn() as conn:
            conn.execute("PRAGMA journal_mode=WAL")
            conn.execute("PRAGMA foreign_keys=ON")
            conn.execute("PRAGMA synchronous=NORMAL")
            self._migrate(conn)

    @contextmanager
    def _conn(self):
        conn = sqlite3.connect(str(self.db_path), timeout=10)
        conn.row_factory = sqlite3.Row
        try:
            yield conn
            conn.commit()
        except:
            conn.rollback(); raise
        finally:
            conn.close()

    def new_session(self, source: str = "cli", model: str = "",
                    user_id: str | None = None, platform: str | None = None) -> str:
        """建立新 session，回傳 session ID（格式：20260518_091523_a1b2c3）。"""
        ts = time.time()
        sid = f"{time.strftime('%Y%m%d_%H%M%S', time.localtime(ts))}_{uuid.uuid4().hex[:6]}"
        with self._conn() as conn:
            conn.execute(
                "INSERT INTO sessions(id, source, model, user_id, platform, started_at) VALUES (?,?,?,?,?,?)",
                (sid, source, model, user_id, platform, ts),
            )
        return sid

    def append_message(self, session_id: str, message: dict) -> None:
        """把一條 message 附加到 session 歷史。"""
        role = message.get("role", "")
        content = message.get("content")
        if isinstance(content, (list, dict)):
            content = json.dumps(content, ensure_ascii=False)
        tool_calls = message.get("tool_calls")
        tc_str = json.dumps(tool_calls, ensure_ascii=False) if tool_calls else None
        with self._conn() as conn:
            conn.execute(
                "INSERT INTO messages(session_id,role,content,tool_call_id,tool_calls,timestamp) VALUES(?,?,?,?,?,?)",
                (session_id, role, content, message.get("tool_call_id"), tc_str, time.time()),
            )
            conn.execute(
                "UPDATE sessions SET message_count = message_count + 1 WHERE id=?", (session_id,)
            )

    def load_session(self, session_id: str) -> list[dict]:
        """載入 session 的完整 message history（可直接傳給 AIAgent）。"""
        with self._conn() as conn:
            rows = conn.execute(
                "SELECT * FROM messages WHERE session_id=? ORDER BY timestamp", (session_id,)
            ).fetchall()
        messages = []
        for row in rows:
            msg: dict = {"role": row["role"]}
            if row["content"] is not None: msg["content"] = row["content"]
            if row["tool_call_id"]: msg["tool_call_id"] = row["tool_call_id"]
            if row["tool_calls"]: msg["tool_calls"] = json.loads(row["tool_calls"])
            messages.append(msg)
        return messages

    def find_session(self, query: str) -> list[dict]:
        """按名稱或 ID 找 session，支援模糊匹配（LIKE）。"""
        with self._conn() as conn:
            rows = conn.execute(
                "SELECT * FROM sessions WHERE id=? OR title LIKE ? ORDER BY started_at DESC LIMIT 10",
                (query, f"%{query}%"),
            ).fetchall()
        return [dict(r) for r in rows]

    def end_session(self, session_id: str, reason: str = "normal", usage: dict | None = None) -> None:
        usage = usage or {}
        with self._conn() as conn:
            conn.execute(
                "UPDATE sessions SET ended_at=?,end_reason=?,input_tokens=?,output_tokens=?,cache_read_tokens=? WHERE id=?",
                (time.time(), reason,
                 usage.get("input_tokens", 0), usage.get("output_tokens", 0),
                 usage.get("cache_read_tokens", 0), session_id),
            )

    def list_sessions(self, limit: int = 20, source: str | None = None) -> list[dict]:
        where = "WHERE source=?" if source else ""
        params = ([source] if source else []) + [limit]
        with self._conn() as conn:
            rows = conn.execute(
                f"SELECT * FROM sessions {where} ORDER BY started_at DESC LIMIT ?", params
            ).fetchall()
        return [dict(r) for r in rows]

MEMORY.md：精選記憶 vs 無限歷史

SQLite 儲存所有對話完整歷史，但每次讀入全部會爆 context window。MEMORY.md 是 agent 主動策劃的精選筆記：只存「下次 session 還會有用」的事實。

系統儲存什麼如何存取上限
SQLite messages所有對話的完整記錄session_search tool（FTS5 搜尋）無上限（磁碟）
MEMORY.md精選的持久事實每次 session 注入 system prompt（frozen snapshot）2,200 chars
USER.md使用者偏好和 profile每次 session 注入 system prompt1,375 chars

MEMORY.md 的格式是自由 Markdown，每個條目以 § 分隔（便於 LLM 用 substring match 定位和更新）：

markdown~/.myagent/memories/MEMORY.md（範例）
使用者的專案是 ~/code/myapi，使用 Rust + Axum + SQLx
§
此機器執行 Ubuntu 22.04，已安裝 Docker 24 和 Podman 4
§
使用者偏好簡潔回覆，不喜歡冗長解釋，程式碼優先
§
已知問題：integration test 需先執行 ./scripts/setup-test-db.sh

Memory Tool：agent 更新記憶的介面

pythontools/memory_tool.py（完整實作）
"""tools/memory_tool.py — Persistent Memory Tool"""
import json
from pathlib import Path
from agent.prompt_builder import get_agent_home, MEMORY_MAX_CHARS, USER_MAX_CHARS
from tools.registry import registry

MEMORY_SCHEMA = {
    "name": "memory",
    "description": (
        "Add, replace, remove or read persistent memory entries (MEMORY.md or USER.md). "
        "Use this to save facts that will still matter next session."
    ),
    "parameters": {
        "type": "object",
        "properties": {
            "action": {
                "type": "string",
                "enum": ["add", "replace", "remove", "read"],
                "description": "add: append. replace: update by old_text. remove: delete by old_text. read: return contents.",
            },
            "target": {"type": "string", "enum": ["memory", "user"], "default": "memory",
                       "description": "'memory' for MEMORY.md, 'user' for USER.md"},
            "content": {"type": "string", "description": "Text to add or replace with"},
            "old_text": {"type": "string", "description": "Unique substring of entry to replace/remove"},
        },
        "required": ["action"],
    },
}


def _get_path(target: str) -> tuple[Path, int]:
    d = get_agent_home() / "memories"
    d.mkdir(parents=True, exist_ok=True)
    return (d / "USER.md", USER_MAX_CHARS) if target == "user" else (d / "MEMORY.md", MEMORY_MAX_CHARS)


def _memory_handler(args: dict) -> str:
    action = args.get("action", "read")
    target = args.get("target", "memory")
    content = args.get("content", "")
    old_text = args.get("old_text", "")
    path, max_chars = _get_path(target)
    current = path.read_text(encoding="utf-8").strip() if path.exists() else ""

    if action == "read":
        return json.dumps({"content": current, "chars": len(current), "max": max_chars})

    if action == "add":
        if not content:
            return json.dumps({"error": "content required"})
        new = (current + "\n§\n" + content) if current else content
        if len(new) > max_chars:
            return json.dumps({"error": f"Memory full ({len(new)}/{max_chars}). Remove old entries first."})
        path.write_text(new, encoding="utf-8")
        return json.dumps({"success": True, "action": "added", "chars": len(new)})

    if action in ("replace", "remove"):
        if not old_text:
            return json.dumps({"error": f"old_text required for {action}"})
        entries = current.split("\n§\n")
        if action == "replace":
            found = False
            for i, e in enumerate(entries):
                if old_text in e:
                    entries[i] = content; found = True; break
            if not found:
                return json.dumps({"error": f"old_text not found: {old_text!r}"})
        else:
            entries = [e for e in entries if old_text not in e]
        path.write_text("\n§\n".join(entries), encoding="utf-8")
        return json.dumps({"success": True, "action": action})

    return json.dumps({"error": f"Unknown action: {action}"})


registry.register(name="memory", toolset="memory", schema=MEMORY_SCHEMA, handler=_memory_handler)
