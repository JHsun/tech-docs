1. block 結構速查

HCL · 所有 block 類型 
# ===== terraform block =====
terraform {
  required_version = ">= 1.11.0, < 2.0.0"

  required_providers {
    aws = { source = "hashicorp/aws", version = "~> 5.0" }
  }

  backend "s3" { /* ... */ }
  encryption  { /* ... */ }   # 1.7+
  experiments = [example]      # 罕用
}

# ===== resource =====
resource "TYPE" "NAME" {
  argument = value
}

# ===== data source =====
data "TYPE" "NAME" {
  argument = value
}

# ===== variable =====
variable "NAME" {
  type        = string
  default     = "value"
  description = "..."
  sensitive   = true       # 標記為敏感
  ephemeral   = true       # 1.11+ 真正不寫入 state
  nullable    = false
  validation  { /* ... */ }
}

# ===== output =====
output "NAME" {
  value       = expression
  description = "..."
  sensitive   = true
  ephemeral   = true       # 1.11+
  deprecated  = "Use X instead."   # 1.11+
}

# ===== local =====
locals {
  name = "value"
  tags = { x = "y" }
}

# ===== module =====
module "NAME" {
  source  = "..."
  version = "~> 1.0"

  variable_a = "value"

  providers = {            # 傳遞 alias provider
    aws = aws.tokyo
  }
}

# ===== provider =====
provider "TYPE" {
  alias  = "ALIAS"          # 區分多個同 provider 設定
  region = "us-east-1"

  for_each = toset([...])   # 1.9+ 動態 provider
}

# ===== ephemeral resource (1.10+) =====
ephemeral "TYPE" "NAME" {
  argument = value
}

# ===== check block =====
check "NAME" {
  data "TYPE" "x" { /* ... */ }
  assert {
    condition     = ...
    error_message = "..."
  }
}

# ===== state 操作 blocks =====
import   { to = ADDR, id = "..." }
moved    { from = ADDR_OLD, to = ADDR_NEW }
removed  { from = ADDR, lifecycle { destroy = false } }

2. 變數型別速查

HCL · 完整型別系統 
# 基本型別
variable "s" { type = string }
variable "n" { type = number }
variable "b" { type = bool }

# any（不建議，避免）
variable "x" { type = any }

# Collection 型別
variable "l1" { type = list(string) }
variable "l2" { type = list(number) }
variable "l3" { type = list(any) }

variable "s1" { type = set(string) }
# set 自動去重、無順序

variable "m1" { type = map(string) }
variable "m2" { type = map(number) }

# Structural 型別
variable "obj" {
  type = object({
    name = string
    age  = number
    tags = optional(list(string), [])  # 1.3+ 預設值
    role = optional(string)
  })
}

variable "tup" {
  type = tuple([string, number, bool])
}

# 複雜巢狀
variable "config" {
  type = map(object({
    cidr     = string
    public   = bool
    az_count = optional(number, 3)
  }))
}

# 動態型別轉換
locals {
  num_str    = tostring(42)         # "42"
  str_num    = tonumber("42")       # 42
  list_set   = tolist(toset([1,2])) # [1, 2]
  set_list   = toset([1, 1, 2])     # [1, 2]
  map_obj    = tomap({a=1, b=2})    # map(number)
}

3. validation 速查

HCL · variable validation 範例 
variable "environment" {
  type = string
  validation {
    condition     = contains(["dev", "staging", "prod"], var.environment)
    error_message = "Environment must be dev, staging, or prod."
  }
}

variable "instance_type" {
  type = string
  validation {
    condition     = can(regex("^t3\\.|^m5\\.", var.instance_type))
    error_message = "Only t3 or m5 instance types allowed."
  }
}

variable "port" {
  type = number
  validation {
    condition     = var.port >= 1024 && var.port <= 65535
    error_message = "Port must be between 1024 and 65535."
  }
}

variable "tags" {
  type = map(string)
  validation {
    condition     = alltrue([for k in ["Owner", "Environment"] : contains(keys(var.tags), k)])
    error_message = "Tags must include Owner and Environment."
  }
}

# OpenTofu 1.9+：validation 可引用其他變數
variable "db_password" {
  type      = string
  sensitive = true
}

variable "db_password_confirm" {
  type      = string
  sensitive = true
  validation {
    condition     = var.db_password_confirm == var.db_password
    error_message = "Passwords don't match."
  }
}

4. 表達式速查

HCL · 表達式速查 
# ===== 條件表達式 =====
locals {
  result = var.is_prod ? "production" : "development"
}

# ===== for 表達式 =====
locals {
  # list → list（轉換）
  upper_names = [for n in var.names : upper(n)]

  # list → list（過濾）
  long_names = [for n in var.names : n if length(n) > 5]

  # list → map
  by_id = { for u in var.users : u.id => u.name }

  # map → map
  upper_keys = { for k, v in var.tags : upper(k) => v }

  # 巢狀 for
  flat = [for env in var.envs : [for s in env.servers : "${env.name}-${s}"]]
  combined = flatten(local.flat)
}

# ===== splat 表達式 =====
locals {
  # 從 list of object 取屬性
  ids = aws_instance.web[*].id     # = [for x in aws_instance.web : x.id]

  # 對 single object 也能用（變成 1-element list）
  vpc_id_list = aws_vpc.main[*].id  # 若 main 是 single resource 也回傳 list
}

# ===== dynamic block =====
resource "aws_security_group" "x" {
  dynamic "ingress" {
    for_each = var.ingress_rules
    content {
      from_port   = ingress.value.port
      to_port     = ingress.value.port
      protocol    = "tcp"
      cidr_blocks = ingress.value.cidrs
    }
  }
}

# ===== 引用 =====
var.NAME                  # 變數
local.NAME                # local
data.TYPE.NAME.attr       # data source
resource.TYPE.NAME.attr   # resource（簡寫：TYPE.NAME.attr）
module.NAME.output_name   # module output
each.key, each.value      # for_each
count.index               # count
self.attr                 # 僅 postcondition / provisioner
terraform.workspace       # 當前 workspace 名

常用函式範例

HCL · 函式實戰 
locals {
  # 字串處理
  bucket_name = lower(replace("MyApp Prod", " ", "-"))   # "myapp-prod"

  # 數值
  half = floor(var.count / 2)

  # collection
  unique_az  = distinct(var.azs)
  first_az   = element(var.azs, 0)
  region     = lookup(var.regions, var.environment, "us-east-1")

  # 編碼
  config_json = jsonencode({
    db_host = aws_db_instance.main.endpoint
    db_port = 5432
  })
  parsed = jsondecode(file("config.json"))

  # 檔案
  user_data = templatefile("${path.module}/user_data.sh.tpl", {
    region = var.region
    env    = var.environment
  })

  # CIDR
  subnet_cidrs = [for i in range(3) : cidrsubnet("10.0.0.0/16", 8, i)]

  # 日期
  timestamp_tag = formatdate("YYYY-MM-DD", timestamp())

  # 安全試錯
  optional_attr = try(var.config.optional_field, "default")
  is_valid_ip   = can(cidrhost(var.cidr, 0))
}

6. Meta-Arguments 速查

HCL · 所有 meta-arguments 
resource "aws_instance" "x" {
  # ===== count =====
  count = 3
  tags  = { Name = "web-${count.index}" }

  # ===== for_each =====（不能與 count 並用）
  # for_each = toset(["a", "b", "c"])
  # tags     = { Name = each.key }

  # for_each with map
  # for_each = var.subnet_config
  # cidr_block = each.value.cidr

  # ===== provider =====（用 alias）
  provider = aws.tokyo

  # ===== depends_on =====（顯式依賴）
  depends_on = [
    aws_security_group.x,
    aws_iam_role.x
  ]

  # ===== lifecycle =====
  lifecycle {
    create_before_destroy = true        # 先建後刪
    prevent_destroy       = true        # 防誤刪
    ignore_changes        = [tags]      # 忽略變更
    replace_triggered_by  = [           # 觸發重建
      aws_launch_template.x.latest_version
    ]

    precondition {                       # 1.2+
      condition     = data.aws_ami.x.architecture == "x86_64"
      error_message = "AMI must be x86_64."
    }

    postcondition {                      # 1.2+
      condition     = self.private_ip != ""
      error_message = "Must have private IP."
    }
  }

  # ===== enabled =====（OpenTofu 1.11+ 獨家）
  enabled = var.create_instance         # true/false 控制是否建立
}

# ===== count 與 for_each 的選擇 =====
# 同質、順序穩定 → count
# 異質、可命名 key → for_each（推薦）

# ===== module 也支援 =====
module "vpc" {
  source = "..."

  count    = var.create_vpc ? 1 : 0
  for_each = var.environments
  providers = { aws = aws.tokyo }
  depends_on = [...]
}

7. Conditions 速查

HCL · 四種驗證機制 
# ===== variable.validation =====
variable "env" {
  type = string
  validation {
    condition     = contains(["dev","staging","prod"], var.env)
    error_message = "..."
  }
}

# ===== precondition =====
resource "aws_instance" "x" {
  lifecycle {
    precondition {
      condition     = data.aws_ami.x.architecture == "x86_64"
      error_message = "..."
    }
  }
}

# ===== postcondition =====
resource "aws_instance" "x" {
  lifecycle {
    postcondition {
      condition     = self.public_dns != ""
      error_message = "..."
    }
  }
}

# ===== check block（只 warning 不擋 apply）=====
check "health" {
  data "http" "endpoint" {
    url = "https://${aws_lb.x.dns_name}/health"
  }
  assert {
    condition     = data.http.endpoint.status_code == 200
    error_message = "Health check failed."
  }
}

四種比較

機制檢查時機失敗行為
variable.validation變數值傳入時立即 error
precondition該物件動作執行前立即 error
postcondition該物件動作執行後立即 error
checkplan/apply 結尾warning（不擋 apply）

8. State 操作 block 速查

HCL · 三種宣告式 state 操作 
# ===== import：納管既有資源 =====
import {
  to = aws_s3_bucket.legacy
  id = "my-existing-bucket"
}

resource "aws_s3_bucket" "legacy" {
  bucket = "my-existing-bucket"
}

# 批次 import
import {
  for_each = var.legacy_buckets
  to       = aws_s3_bucket.legacy[each.value]

# tofu plan -generate-config-out=generated.tf

# 改名
moved {
  from = aws_instance.web
  to   = aws_instance.web_primary
}

moved { from = aws_instance.web[0], to = aws_instance.web["a"] }
moved { from = aws_instance.web[1], to = aws_instance.web["b"] }
moved { from = aws_instance.web[2], to = aws_instance.web["c"] }

  from = aws_vpc.main

9. Ephemeral 速查（OpenTofu 1.11+）

HCL · 4 類 ephemerality 
# ===== ① ephemeral resource =====
ephemeral "openbao_kv_v2_secret" "db" {
  mount = "kv"
  name  = "db/master"
}

# 用法（只能在 ephemeral context）
provider "aws" {
  secret_key = ephemeral.openbao_kv_v2_secret.db.data["secret_key"]
}

# ===== ② ephemeral variable =====
variable "session_token" {
  type      = string
  ephemeral = true
}

# ===== ③ ephemeral output =====
output "db_secret" {
  value     = ephemeral.openbao_kv_v2_secret.db.data["password"]
  ephemeral = true
  sensitive = true
}

# ===== ④ write-only attribute =====
resource "aws_db_instance" "main" {
  password_wo         = ephemeral.openbao_kv_v2_secret.db.data["password"]
  password_wo_version = 1   # 改 version 才會更新
}

# 註：password_wo 只在「設定變更時」存在，平時不在 state

ephemeral 使用位置限制

位置能用 ephemeral？
provider block 屬性✅
provisioner / connection block✅
另一個 ephemeral 資源/輸出/變數✅
local 值（傳染性）✅
resource 的 write-only 屬性✅
resource 的一般屬性❌
非 ephemeral output❌
data source 的 arguments❌

10. version 約束速查

運算子意義範例
= 1.11.6 或 1.11.6精確等於只允許 1.11.6
!= 1.11.0不等於排除 1.11.0
>= 1.6.0大於等於1.6.0+
< 2.0.0小於1.x.x
~> 1.11悲觀約束（最右一位可變）1.11.x、1.12.x... 但不到 2.0
~> 1.11.0悲觀約束（patch 可變）1.11.x 但不到 1.12.0
>= 1.6, < 2.0範圍組合1.6.x ~ 1.x.x

實戰策略

HCL · root vs reusable module 策略 
# Root module（執行 apply 的地方）：寫上下限
terraform {
  required_version = ">= 1.11.0, < 2.0.0"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.30"        # 鎖在 minor 系列
    }
  }
}

# Reusable module（給別人用）：只寫下限
terraform {
  required_version = ">= 1.6.0"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = ">= 5.0"          # 不鎖上限，給彈性
    }
  }
}

11. 字串模板速查

HCL · 字串模板語法 
locals {
  # 內插 ${ ... }
  greeting = "Hello, ${var.name}!"
  json     = "{\"name\": \"${var.name}\"}"

  # heredoc
  config = <<-EOT
    region = ${var.region}
    env    = ${var.environment}
  EOT

  # 條件
  msg = "Server is ${var.is_active ? "running" : "stopped"}"

  # for（在字串模板中）
  list_str = "Items: %{for item in var.items}${item}, %{endfor}"

  # if
  status = "%{if var.enabled}enabled%{else}disabled%{endif}"

  # 跳脫 $
  literal = "Cost: $$100"   # 輸出 "Cost: $100"

  # %% 跳脫
  format_str = "Progress: 50%%"   # 輸出 "Progress: 50%"

  # 修剪 whitespace（- 號）
  trimmed = <<-EOT
    %{ for s in var.servers ~}
    server: ${s}
    %{ endfor ~}
  EOT
}
